cd ~/davartapp

# остановить всё и удалить тома
docker compose -f infra/docker-compose.staging.yml --env-file infra/.env.staging down -v

# на всякий случай убрать висящие тома проекта
docker volume ls
docker volume prune -f

# поднять заново
docker compose -f infra/docker-compose.staging.yml --env-file infra/.env.staging up -d --build

# логи api в реальном времени (Ctrl+C чтобы выйти)
docker compose -f infra/docker-compose.staging.yml --env-file infra/.env.staging logs -f api
